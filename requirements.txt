# GTS Vaccination Tracking pipeline — runtime dependencies.
#
# Lower bounds are the versions the pipeline is developed and tested against.
# Upper bounds stop the next MAJOR release from breaking a deploy without
# warning — geopandas 1.x and numpy 2.x in particular changed APIs this code
# relies on.
#
# Every one of these installs from a pre-built wheel on Streamlit Community
# Cloud (Linux/x86-64, Python 3.11+). No system packages are required: the
# geospatial stack below bundles its own GEOS, PROJ and GDAL, which is why
# there is no `packages.txt` in this repo. See DEPLOYMENT.md.

# ---- app ------------------------------------------------------------------
streamlit>=1.36,<2

# ---- data -----------------------------------------------------------------
pandas>=2.1,<3
numpy>=1.26,<3
pyarrow>=15                  # parquet for the day's gridded target area

# ---- geospatial -----------------------------------------------------------
# geopandas 1.x defaults to pyogrio, which ships GDAL inside its wheel.
# fiona is deliberately NOT listed: it is the older engine, needs a system
# GDAL, and is the usual cause of a failed geospatial build on a hosted
# runner. shapely 2.x and pyproj bundle GEOS and PROJ respectively.
geopandas>=1.0,<2
shapely>=2.0,<3
pyogrio>=0.8
pyproj>=3.6

# ---- charts and maps ------------------------------------------------------
matplotlib>=3.8,<4
pillow>=10

# ---- report output --------------------------------------------------------
python-docx>=1.1
python-pptx>=1.0
XlsxWriter>=3.1
openpyxl>=3.1
